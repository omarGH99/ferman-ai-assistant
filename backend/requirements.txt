# Pinned deliberately. `gcloud run deploy --source .` rebuilds the image from
# scratch, so unpinned versions meant every deploy silently pulled whatever was
# newest — a transformers or pydantic major release could break production with
# no code change on our side. These are the versions the app is tested against.
#
# To upgrade: bump here, run `python test_app.py` from backend/, then deploy and check /health.
#
# torch: `==2.13.0` also matches the `2.13.0+cpu` wheel the Dockerfile installs
# first from PyTorch's CPU index, so pip treats it as satisfied and never pulls
# the multi-GB CUDA build. Keeping it listed means a plain
# `pip install -r requirements.txt` still works for local development.
fastapi==0.140.0
uvicorn[standard]==0.51.0
transformers==5.14.1
torch==2.13.0
huggingface_hub==1.24.0
pydantic==2.13.4
sentencepiece==0.2.2
PyJWT==2.13.0
psycopg[binary]==3.3.4
